import json import re from copy import deepcopy LINK_SUFFIX_RE = re.compile(r"\s*\(Link\s*(\d+)\)\s*$", re.IGNORECASE) def base_name_and_index(name: str): """ "Match ... (Link 3)" -> ("Match ...", 3) "Match ..." -> ("Match ...", 1) """ name = (name or "").strip() m = LINK_SUFFIX_RE.search(name) if m: idx = int(m.group(1)) base = LINK_SUFFIX_RE.sub("", name).strip() return base, max(idx, 1) return name, 1 def make_child_channel(parent: dict, original_channel: dict, idx: int): """ Tạo 1 kênh con từ channel gốc (Link idx) - name: "Link {idx}" - type: "single" - giữ nguyên sources / image / display... """ child = deepcopy(original_channel) # Kênh con chỉ hiển thị là Link 1/2/3... child["name"] = f"Link {idx}" child["type"] = "single" # Label/time không bắt buộc cho con, nhưng giữ lại cũng ok. # Nếu muốn gọn thì có thể xóa labels/description ở con: # child.pop("labels", None) # child.pop("description", None) # ID con để không trùng base_id = original_channel.get("id", "ch") child["id"] = f"{base_id}-link{idx}" return child def merge_channels_into_parent_group(channels: list) -> list: """ Gộp các channel cùng base name thành 1 channel cha type='group' có field 'channels' con. """ if not channels: return channels # group theo base name bucket = {} order_keys = [] # giữ thứ tự xuất hiện for ch in channels: base, idx = base_name_and_index(ch.get("name", "")) key = base.lower() if key not in bucket: bucket[key] = [] order_keys.append(key) bucket[key].append((idx, ch, base)) out = [] for key in order_keys: items = bucket[key] items.sort(key=lambda x: x[0]) # Link 1..N # chỉ 1 link thì giữ nguyên if len(items) == 1: out.append(items[0][1]) continue # chọn kênh cha = bản Link 1 (nếu có), hoặc idx nhỏ nhất idx0, ch0, base_name = items[0] parent = deepcopy(ch0) # sửa name cha về base (bỏ "(Link N)") parent["name"] = base_name # cha là group để bấm vào bung con parent["type"] = "group" # remove sources của cha (quan trọng): # cha chỉ là folder, không nên có sources để tránh app hiểu là stream luôn parent.pop("sources", None) # đảm bảo display vẫn như cũ (mày đang dùng text-below) parent.setdefault("display", "text-below") # tạo list kênh con children = [] for idx, ch, _base in items: children.append(make_child_channel(parent, ch, idx)) parent["channels"] = children out.append(parent) return out def apply_merge_to_data(data: dict) -> dict: data = deepcopy(data) for g in data.get("groups", []): if isinstance(g, dict) and isinstance(g.get("channels"), list): g["channels"] = merge_channels_into_parent_group(g["channels"]) return data if __name__ == "__main__": INP = "buncha.json" OUT = "buncha_merged.json" with open(INP, "r", encoding="utf-8") as f: data = json.load(f) data2 = apply_merge_to_data(data) with open(OUT, "w", encoding="utf-8") as f: json.dump(data2, f, ensure_ascii=False, separators=(",", ":")) print(f"OK: wrote {OUT}")
